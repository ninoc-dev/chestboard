window.addEventListener("scroll", () => {
    const {top} = logo.getBoundingClientRect();
    
    if (top < 10 && y === null) {
        y = window.pageYOffset;
        console.log(y);
        logo.style.fontSize = getComputedStyle(logo).fontSize;
        animate(logo, {fontSize: ['4.5rem', '2rem']});
        logo.classList.add("sticky");
        
        if (linki.length > 1) {
            linki[1].style.marginRight = '28rem';
        }
    }
    else if (window.pageYOffset < y) {
        logo.classList.remove("sticky");
        animate(logo, {fontSize: '4.5rem'});
        
        if (linki.length > 1) {
            linki[1].style.marginRight = '';
        }

        y = null;
    }
});


4.5rem = 72px
2rem = 16px

72:16 = 9:2 = 4.5:1

press('.razvoj button', (btn) => {
    const section = document.querySelector('.razvoj')
    const index = btn.dataset.index;
    const hidSec = document.querySelector(`.hidRaz${index}`);
    const hidCont = hidSec.querySelector('.hidVsebina');
    const closeBtn = hidSec.querySelector('.closeBtn');

    if (!hidSec) {
        console.error(`Element .hidRaz${index} ni najden`);
        return () => {};
    }       

    const overlay = document.createElement('div');
    overlay.classList.add('overlay');
    overlay.style.display = 'flex'

    const zaOkno = document.createElement('div');
    zaOkno.classList.add('zaOkno');

    overlay.appendChild(zaOkno)
    overlay.appendChild(hidSec);
    section.appendChild(overlay);
    body.style.overflow = 'hidden';
    animate(logo, {opacity: 0}, {duration: 0.1})
    animate(linki, {opacity: 0}, {duration: 0.1})

    animate(overlay, {opacity: 1})
    animate(zaOkno, {opacity: 1, scale:2}, {delay: 0.1})
    
    const odpiranje = [
        [hidSec.style.display = 'block'],
        [hidCont.style.display = 'block'],
        [hidSec, {opacity: 1}],
        [zaOkno, {opacity: 0}],
        [zaOkno.remove()],
        [hidSec, {maxHeight: ['100vh', '96vh']}, {duration: 0.1}],
        ['.hidVsebina > *', {opacity: 1, x: [10, 0]}, {delay: stagger(0.05), at: 0.1}],
        // [colseBtn, {opacity: 1, rotate: [0, 180]}, {duration: 0.5, ease: 'easeOut', at: 1}]
    ]
    animate(odpiranje)

    press(closeBtn, (element) => {
        animate(element, {rotate: 360})

        return () => {}
    })

    return () => {}
})
